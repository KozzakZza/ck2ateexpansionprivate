mex_crisis_end = {
	name = CB_NAME_MEX_CRISIS_END
	war_name = WAR_NAME_MEX_CRISIS_END
	sprite = 17
	truce_days = 1840
	is_permanent = yes
	max_attacker_battle_score = 20 #you should at least have to siege down something, and the wars should have the potential to be longer and costlier
	allowed_to_target_tributaries = yes
	
	can_use = {
		ROOT = {
		    OR = {
			    has_landed_title = k_mexico
				has_landed_title = k_west_mexico
				has_landed_title = k_veracruz_mexico
			}
		}
		FROM = {
			OR = {
			    has_landed_title = k_mexico
				has_landed_title = k_west_mexico
				has_landed_title = k_veracruz_mexico
			}
		}
		NAND = {
		    ROOT = { has_landed_title = k_west_mexico }
			FROM = { has_landed_title = k_veracruz_mexico }
		}
		NAND = {
		    ROOT = { has_landed_title = k_veracruz_mexico }
			FROM = { has_landed_title = k_west_mexico }
		}
	}
	
	is_valid = {
		ROOT = {
			independent = yes
		}
	}
	
	on_success = {
		if = {
		   limit = { any_attacker = { has_landed_title = k_mexico } }
		   ROOT = {
		      gain_title = k_west_mexico
		   }
		   activate_title = { status = no title = k_west_mexico }
		   k_west_mexico = { destroy_landed_title = THIS }
		}
		if = {
		   limit = { any_attacker = { has_landed_title = k_west_mexico } }
		   ROOT = {
		      gain_title = k_mexico
			  gain_title = d_mexico
			  gain_title = c_mexico
		   }
		   k_mexico = {
			    copy_title_laws = k_west_mexico
			    copy_title_history = k_west_mexico
		   }
		   activate_title = { status = no title = k_west_mexico }
		   k_west_mexico = { destroy_landed_title = THIS }
		}
		if = {
		   limit = { any_attacker = { has_landed_title = k_veracruz_mexico } }
		   ROOT = {
		      gain_title = k_mexico
			  gain_title = d_mexico
			  gain_title = c_mexico
		   }
		   k_mexico = {
			    copy_title_laws = k_veracruz_mexico
			    copy_title_history = k_veracruz_mexico
		   }
		   activate_title = { status = no title = k_veracruz_mexico }
		   k_veracruz_mexico = { destroy_landed_title = THIS }
		}
	}

	on_fail = {
		ROOT = { 
			prestige = -400
			cash = -100
		}
	}

	on_reverse_demand = {
		ROOT = {
		    prestige = -1000
			cash = -500
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}
