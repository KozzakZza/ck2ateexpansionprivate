narrative_event = { #Collapse of Kingdom of Canada
	id = canada_new.2
	title = EVTNAME_canada_new.2
	desc = EVTDESC_canada_new.2
	major = yes
	
	mean_time_to_happen = {
		days = 1
	}
	
	only_playable = yes
	
	trigger = {
	      k_canada = {
		     has_holder = no
		  }
		  NOT = { has_global_flag = k_canada_collapse }
	}
	
	immediate = {
		activate_title = { status = no title = d_canada_royal_guard }
        d_canada_royal_guard = { destroy_landed_title = THIS }
		set_global_flag = k_canada_collapse
	}
	
	option = {
		name = EVTOPT_canada_new.2
	}
}
