namespace = edokko

character_event = {
	id = edokko.1
	desc = EVTDESCedokko.1
	
	trigger = {
	    has_landed_title = e_edokko
		has_global_flag = ihaveahighground
		NOT = { religion = cetic }
	    NOT = { has_global_flag = japanese_take_sacramento }
		NOT = { has_global_flag = californian_emperor_puppetised }
		NOT = { has_global_flag = edokko_can_vassalize_emperor_of_california }
		OR = {
		   completely_controls = c_solano
		   completely_controls = c_yolo
		   completely_controls = c_san_joaquin
		   completely_controls = c_dorado
		   completely_controls = c_tahoe
		   completely_controls = c_yuba
		}
		NOT = {
		   	e_california = {
				has_holder = no
			}
		}
	}
	
	picture = GFX_evt_colonial_japan
	
	#is_triggered_only = yes
	
	option = {
		name = EVTOPTAedokko.1
		custom_tooltip = { 
			text = edokko_tooltip_1
			hidden_tooltip = {
			   set_global_flag = edokko_can_vassalize_emperor_of_california
			}
		}
	}
}

narrative_event = {
	id = edokko.2
	title = EVTTITLEedokko.2
	desc = EVTDESCedokko.2
	
	picture = GFX_evt_colonial_japan
	
	is_triggered_only = yes
	
	option = {
		name = EVTOPTAedokko.2
		hidden_tooltip = {
	      k_puppet_california = {
			  show_scope_change = no
			  grant_title = ROOT
			  hidden_tooltip = { make_primary_title = yes }
			  copy_title_laws = e_california
			  copy_title_history = e_california
		   }
		   	k_puppet_california = {
			   holder_scope = {
	               set_defacto_liege = e_edokko:holder_scope
			    }  
	        }
			if = {
			  limit = {
			     k_puppet_california = {
				    holder_scope = {
				       has_landed_title = k_sacramento
				    }
				 }
			  }
			k_sacramento = {
			   holder_scope = {
	               set_defacto_liege = e_edokko:holder_scope
			    }  
	        }
			}
			e_california = {
			   show_scope_change = no
			   destroy_landed_title = THIS
		    }
		   	cetic = {
			    set_religion_head = k_puppet_california
		    }
		   set_global_flag = californian_emperor_puppetised
		   clr_global_flag = edokko_can_vassalize_emperor_of_california
		   any_character = {
		       limit = {
			       ai = no
			   }
			   narrative_event = { id = edokko.4 }
		   }
		}
	}
}
