namespace = URA

character_event = {
	id = URA.1
	desc = EVTDESC_URA.1_desc
	picture = GFX_evt_americanist_intervention
	
	trigger = {
        has_landed_title = k_united_republics
		NOT = { has_character_modifier = has_ura_president_term }
    }

    mean_time_to_happen = {
        days = 1
    }
	
	option = {
		name = EVTOPT_URA.1
		add_trait="ura_president"
		current_heir = { save_event_target_as = new_ura_president }
		event_target:new_ura_president = { 
		   gain_title = k_united_republics
		   gain_title = c_suffolk
		   add_trait="ura_president"
		   add_character_modifier = {
			  name = has_ura_president_term
			  years = 7
			  hidden = yes
		   }
		}
		if = {
		   limit = { ai = no }
		   set_player_character = event_target:new_ura_president
		}
		clear_event_target = new_ura_president
	}
}
